cmake_minimum_required(VERSION 3.12)
project(RoundRobin LANGUAGES CXX)
enable_testing()
add_executable(core_test tests/test.cpp)
target_include_directories(core_test PRIVATE include)
set_target_properties(core_test PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(core_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME core_test COMMAND core_test)
