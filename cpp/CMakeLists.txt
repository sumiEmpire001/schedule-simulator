cmake_minimum_required(VERSION 3.12)
project(RoundRobin LANGUAGES CXX)
enable_testing()
add_executable(core_test tests/test.cpp)
target_include_directories(core_test PRIVATE include)
set_target_properties(core_test PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
add_executable(embedded_core_test embedded_core.cpp)
target_compile_definitions(embedded_core_test PRIVATE EMBEDDED_TEST=1)
set_target_properties(embedded_core_test PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(core_test PRIVATE -Wall -Wextra -Werror -pedantic)
 target_compile_options(embedded_core_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME core_test COMMAND core_test)
add_test(NAME embedded_core_test COMMAND embedded_core_test)
